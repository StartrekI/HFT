            function updateProgress() {

                checkedCount = document.querySelectorAll('.section-check:checked').length;

                const percentage = totalCheckboxes > 0 ? Math.round((checkedCount / totalCheckboxes) * 100) : 0;

                if (progressText) {

                    progressText.textContent = percentage + '%';

                }

                // Save to localStorage

                localStorage.setItem('hft-roadmap-progress', JSON.stringify(getCheckedItems()));

            }

                const checked = [];

                checkboxes.forEach(function(cb) {

                    if (cb.checked) {

                        checked.push(cb.dataset.section);

                    }

                });

                return checked;

            }


            function loadProgress() {

                const saved = localStorage.getItem('hft-roadmap-progress');

                if (saved) {

                    const checkedItems = JSON.parse(saved);

                        if (checkedItems.includes(cb.dataset.section)) {

                            cb.checked = true;

                        }

                    });

                }

                updateProgress();

            }
